1.Получить токен
TOKEN="$(docker-compose -f docker-compose.ci.yml exec -T web \
  python3 -c "from service.security import create_access_token; print(create_access_token(user_id=1, expires_minutes=120))")"
AUTH_HEADER="Authorization: Bearer $TOKEN"


2.Prepare
FILE="/home/vadim/Downloads/video5204062485010747752.mp4"
FILENAME="$(basename "$FILE")"
FILESIZE="$(stat -c%s "$FILE")"

curl -sS -X POST "http://localhost:8000/api/v1/videos/upload/prepare" \
  -H "$AUTH_HEADER" \
  -H "Content-Type: application/json" \
  -d "{\"title\":\"test video\",\"filename\":\"$FILENAME\",\"file_size\":$FILESIZE}" | jq .


3.Upload direct
VIDEO_ID=1
curl -sS -X POST "http://localhost:8000/api/v1/videos/${VIDEO_ID}/upload/direct" \
  -H "$AUTH_HEADER" \
  -F "file=@$FILE" | jq .


4.Complete
UPLOAD_ID="<<<из prepare>>>"

curl -sS -X POST "http://localhost:8000/api/v1/videos/${VIDEO_ID}/upload/complete" \
  -H "$AUTH_HEADER" \
  -H "Content-Type: application/json" \
  -d "{\"upload_id\":\"$UPLOAD_ID\"}" | jq .
